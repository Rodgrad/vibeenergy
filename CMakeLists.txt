cmake_minimum_required(VERSION 3.20)
project(EnergyTLK VERSION 1.0.1 LANGUAGES CXX)
set(CMAKE_AUTOMOC ON)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Qml Gui Quick Multimedia DBus)
find_package(LayerShellQt REQUIRED)
find_package(KF6WindowSystem REQUIRED)
find_package(KF6StatusNotifierItem REQUIRED)
add_executable(energytlk-instance native/instance.cpp)
target_link_libraries(energytlk-instance PRIVATE Qt6::Core Qt6::DBus)
add_executable(energytlk-splash native/splash.cpp)
target_link_libraries(energytlk-splash PRIVATE Qt6::Gui Qt6::Qml Qt6::Quick Qt6::DBus LayerShellQt::Interface)
add_library(energytlk-window-alpha SHARED native/windowalpha.cpp)
target_link_libraries(energytlk-window-alpha PRIVATE Qt6::Gui Qt6::Quick)
add_library(energylocal MODULE native/plugin.cpp native/plasmafield.cpp native/plasmafield.h native/turboaudiomixer.cpp native/turboaudiomixer.h)
target_link_libraries(energylocal PRIVATE Qt6::Core Qt6::Qml Qt6::Gui Qt6::Quick Qt6::Multimedia Qt6::DBus KF6::WindowSystem KF6::StatusNotifierItem)
set_target_properties(energylocal PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/EnergyLocal")
configure_file(native/qmldir "${CMAKE_BINARY_DIR}/EnergyLocal/qmldir" COPYONLY)
include(CTest)
if(BUILD_TESTING)
